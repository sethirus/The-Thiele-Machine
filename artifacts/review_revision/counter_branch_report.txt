counter_branch_result =
fun (s : VMState.VMState) (target : nat) (equal : bool) =>
{|
  VMState.vm_graph := VMState.vm_graph s;
  VMState.vm_csrs :=
    if equal
    then VMState.vm_csrs s
    else VMState.csr_set_err (VMState.vm_csrs s) 1;
  VMState.vm_regs := VMState.vm_regs s;
  VMState.vm_mem := VMState.vm_mem s;
  VMState.vm_pc := target;
  VMState.vm_mu := VMState.vm_mu s + 1;
  VMState.vm_mu_tensor := VMState.vm_mu_tensor s;
  VMState.vm_err := if equal then VMState.vm_err s else true;
  VMState.vm_logic_acc := VMState.vm_logic_acc s;
  VMState.vm_mstatus := VMState.vm_mstatus s;
  VMState.vm_witness := VMState.vm_witness s;
  VMState.vm_certified := VMState.vm_certified s
|}
     : VMState.VMState -> nat -> bool -> VMState.VMState

counter_iteration_result =
fun (s : VMState.VMState) (u v : nat) =>
{|
  VMState.vm_graph := VMState.vm_graph s;
  VMState.vm_csrs := VMState.csr_set_err (VMState.vm_csrs s) 1;
  VMState.vm_regs := VMState.vm_regs s;
  VMState.vm_mem := VMState.vm_mem s;
  VMState.vm_pc := 0;
  VMState.vm_mu := VMState.vm_mu s + 1;
  VMState.vm_mu_tensor := VMState.vm_mu_tensor s;
  VMState.vm_err := true;
  VMState.vm_logic_acc := VMState.vm_logic_acc s;
  VMState.vm_mstatus := VMState.vm_mstatus s;
  VMState.vm_witness := VMUnboundedCounterAccess.counter_witness u (S v);
  VMState.vm_certified := VMState.vm_certified s
|}
     : VMState.VMState -> nat -> nat -> VMState.VMState

counter_rounds =
fix counter_rounds (s : VMState.VMState) (u v n : nat) {struct n} :
    VMState.VMState :=
  match n with
  | 0 => s
  | S k => counter_rounds (counter_iteration_result s u v) u (S v) k
  end
     : VMState.VMState -> nat -> nat -> nat -> VMState.VMState

SimulationProof.run_vm =
fix run_vm
  (fuel : nat) (trace : list VMStep.VMStep.vm_instruction)
  (s : VMState.VMState) {struct fuel} : VMState.VMState :=
  match fuel with
  | 0 => s
  | S fuel' =>
      match List.nth_error trace (VMState.vm_pc s) with
      | Some instr => run_vm fuel' trace (SimulationProof.vm_apply s instr)
      | None => s
      end
  end
     : nat ->
       list VMStep.VMStep.vm_instruction ->
       VMState.VMState -> VMState.VMState
