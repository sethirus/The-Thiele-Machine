fix eval_linear_action
  (k : Syntax.Kind) (old : Semantics.RegsT)
  (a : Syntax.ActionT Syntax.type k) {struct a} :
    option (Semantics.UpdatesT * Syntax.type k) :=
  match a with
  | @Syntax.Let_ _ _ lretT' e cont =>
      eval_linear_action k old (cont (Semantics.evalExpr e))
  | Syntax.ReadReg r kind cont =>
      match action_read old r kind with
      | Some v => eval_linear_action k old (cont v)
      | None => None
      end
  | @Syntax.WriteReg _ _ r k0 e cont =>
      match eval_linear_action k old cont with
      | Some (updates, ret) =>
          match
            FMap.M.find
              (elt:={x : Syntax.FullKind & Syntax.fullType Syntax.type x}) r
              updates
          with
          | Some _ => None
          | None =>
              Some
                (FMap.M.add r
                   (existT (Syntax.fullType Syntax.type) k0
                      (Semantics.evalExpr e)) updates, ret)
          end
      | None => None
      end
  | Syntax.Assert_ p cont =>
      if Semantics.evalExpr p then eval_linear_action k old cont else None
  | Syntax.Displ _ cont => eval_linear_action k old cont
  | Syntax.Return e =>
      Some
        (FMap.M.empty {x : Syntax.FullKind & Syntax.fullType Syntax.type x},
         Semantics.evalExpr e)
  | _ => None
  end
     : forall k : Syntax.Kind,
       Semantics.RegsT ->
       Syntax.ActionT Syntax.type k ->
       option (Semantics.UpdatesT * Syntax.type k)

fix linear_action
  (k : Syntax.Kind) (a : Syntax.ActionT Syntax.type k) {struct a} : Prop :=
  match a with
  | @Syntax.Let_ _ _ lretT' _ cont =>
      forall v : Syntax.fullType Syntax.type lretT', linear_action k (cont v)
  | Syntax.ReadReg _ k0 cont =>
      match

           Prop)

          fun

                    Syntax.ActionT Syntax.type k =>
          forall v : Syntax.fullType Syntax.type (Syntax.SyntaxKind k1),
          linear_action k (cont0 v)
      | @Syntax.NativeKind t c =>
          fun
            _ : Syntax.fullType Syntax.type (Syntax.NativeKind c) ->
                Syntax.ActionT Syntax.type k => False
      end cont
  | Syntax.WriteReg _ _ cont | Syntax.Assert_ _ cont | Syntax.Displ _ cont =>
      linear_action k cont
  | Syntax.Return _ => True
  | _ => False
  end
     : forall k : Syntax.Kind, Syntax.ActionT Syntax.type k -> Prop
