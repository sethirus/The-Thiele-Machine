SoundStructuralShortcut (fuel : nat)
(trace : list VMStep.VMStep.vm_instruction) (s_init : VMState.VMState)
  : Type := Build_SoundStructuralShortcut
  { shortcut_decoder : NoFreeInsight.NoFreeInsight.receipt_decoder
                         VMStep.VMStep.vm_instruction;
    shortcut_obs_fn : VMState.VMState -> list VMStep.VMStep.vm_instruction;
    shortcut_repr_obs_fn : VMState.VMState ->
                           list VMStep.VMStep.vm_instruction;
    shortcut_obs_eqb : list VMStep.VMStep.vm_instruction ->
                       list VMStep.VMStep.vm_instruction -> bool;
    shortcut_omega_prior : list VMState.VMState;
    shortcut_omega_posterior : list VMState.VMState;
    shortcut_tree : MuShannonBridge.DecisionTree;
    shortcut_obs_eqb_spec : forall o1 o2 : list VMStep.VMStep.vm_instruction,
                            shortcut_obs_eqb o1 o2 = true <-> o1 = o2;
    shortcut_strict_narrowing : InformationGainToStrengthening.is_strict_subset
                                  shortcut_omega_posterior
                                  shortcut_omega_prior;
    shortcut_distinguishing_witness : exists s_witness : VMState.VMState,
                                        List.In s_witness
                                          shortcut_omega_prior /\
                                        ~
                                        List.In s_witness
                                          shortcut_omega_posterior /\
                                        InformationGainToStrengthening.observation_distinguishes
                                          shortcut_obs_fn s_witness
                                          shortcut_omega_posterior;
    shortcut_initial_uncertified : VMState.csr_cert_addr
                                     (VMState.vm_csrs s_init) = 0;
    shortcut_certified_posterior : NoFreeInsight.NoFreeInsight.Certified
                                     (SimulationProof.run_vm fuel trace
                                        s_init) shortcut_decoder
                                     (InformationGainToStrengthening.omega_predicate
                                        shortcut_obs_eqb
                                        shortcut_omega_posterior
                                        shortcut_obs_fn) trace;
    shortcut_tree_realized : MuShannonBridge.decision_tree_realized_by_trace
                               fuel trace s_init shortcut_tree;
    shortcut_posterior_nonempty : MuShannonBridge.feasible_size
                                    shortcut_omega_posterior > 0;
    shortcut_representatives : MuShannonBridge.PosteriorRepresentativeReduction
                                 shortcut_repr_obs_fn shortcut_tree
                                 shortcut_omega_prior
                                 shortcut_omega_posterior }.

  shortcut_representatives

WFDrivenPrecondition =
fun (ks : Abstraction.KamiSnapshot) (i : VMStep.VMStep.vm_instruction) =>
match i with
| VMStep.VMStep.instr_pnew region _ =>
    length (VMState.normalize_region region) > 0 /\
    (forall n : nat,
     Abstraction.snap_module_tensors ks (Abstraction.snap_pt_next_id ks) n =
     0)
| VMStep.VMStep.instr_psplit module _ _ _ =>
    pt_well_formed ks /\
    PeanoNat.Nat.modulo module ThieleTypes.PTableSz <
    Abstraction.snap_pt_next_id ks /\
    Abstraction.snap_pt_sizes ks
      (PeanoNat.Nat.modulo module ThieleTypes.PTableSz) >= 2 /\
    Abstraction.snap_pt_sizes ks (Abstraction.snap_pt_next_id ks) = 0 /\
    Abstraction.snap_pt_sizes ks (S (Abstraction.snap_pt_next_id ks)) = 0 /\
    S (S (Abstraction.snap_pt_next_id ks)) <= ThieleTypes.PTableSz /\
    (forall n : nat,
     Abstraction.snap_module_tensors ks (Abstraction.snap_pt_next_id ks) n =
     0) /\
    (forall n : nat,
     Abstraction.snap_module_tensors ks (S (Abstraction.snap_pt_next_id ks))
       n = 0)
| VMStep.VMStep.instr_pmerge m1 m2 _ =>
    pt_well_formed ks /\
    PeanoNat.Nat.modulo m1 ThieleTypes.PTableSz <
    Abstraction.snap_pt_next_id ks /\
    PeanoNat.Nat.modulo m2 ThieleTypes.PTableSz <
    Abstraction.snap_pt_next_id ks /\
    PeanoNat.Nat.modulo m1 ThieleTypes.PTableSz <>
    PeanoNat.Nat.modulo m2 ThieleTypes.PTableSz /\
    Abstraction.snap_pt_sizes ks
      (PeanoNat.Nat.modulo m1 ThieleTypes.PTableSz) > 0 /\
    Abstraction.snap_pt_sizes ks
      (PeanoNat.Nat.modulo m2 ThieleTypes.PTableSz) > 0 /\
    Abstraction.snap_pt_sizes ks (Abstraction.snap_pt_next_id ks) = 0 /\
    S (Abstraction.snap_pt_next_id ks) <= ThieleTypes.PTableSz /\
    (forall n : nat,
     Abstraction.snap_module_tensors ks (Abstraction.snap_pt_next_id ks) n =
     0)
| VMStep.VMStep.instr_lassert freg _ _ flen _ =>
    flen = VMStep.VMStep.lassert_hw_flen (Abstraction.abs_phase1 ks) freg
| VMStep.VMStep.instr_call _ _ =>
    EmbedStep_WF.WellFormedSnapshot ks /\
    Abstraction.snap_pc ks < VMState.MEM_SIZE
| VMStep.VMStep.instr_ret _ => EmbedStep_WF.WellFormedSnapshot ks
| VMStep.VMStep.instr_chsh_trial x y a b _ =>
    VMStep.VMStep.chsh_bits_ok x y a b = true
| VMStep.VMStep.instr_tensor_set _ i0 j _ _ =>
    VMStep.VMStep.tensor_indices_ok i0 j = true
| VMStep.VMStep.instr_tensor_get _ mid i0 j _ =>
    VMStep.VMStep.tensor_indices_ok i0 j = true /\
    VMState.graph_lookup (Abstraction.snap_full_graph ks) mid <> None
| VMStep.VMStep.instr_morph _ src_mod dst_mod _ _ =>
    extended_hw_invariant ks /\
    src_mod < Abstraction.snap_pt_next_id ks /\
    dst_mod < Abstraction.snap_pt_next_id ks /\
    negb (PeanoNat.Nat.eqb (Abstraction.snap_pt_sizes ks src_mod) 0) = true /\
    negb (PeanoNat.Nat.eqb (Abstraction.snap_pt_sizes ks dst_mod) 0) = true
| VMStep.VMStep.instr_morph_id _ module _ =>
    coupling_zero_empty (Abstraction.snap_rich_state ks) /\
    module < Abstraction.snap_pt_next_id ks /\
    negb (PeanoNat.Nat.eqb (Abstraction.snap_pt_sizes ks module) 0) = true
| VMStep.VMStep.instr_morph_delete _ _ |
  VMStep.VMStep.instr_morph_assert _ _ _ _ =>
    morph_table_wf (Abstraction.snap_rich_state ks)
| VMStep.VMStep.instr_compose _ _ _ _ |
  VMStep.VMStep.instr_morph_tensor _ _ _ _ |
  VMStep.VMStep.instr_morph_get _ _ _ _ => extended_hw_invariant ks
| _ => True
end
     : Abstraction.KamiSnapshot -> VMStep.VMStep.vm_instruction -> Prop

WFDrivenRun =
fix WFDrivenRun
  (fuel : nat) (trace : list VMStep.VMStep.vm_instruction)
  (ks : Abstraction.KamiSnapshot) {struct fuel} : Prop :=
  match fuel with
  | 0 => True
  | S fuel' =>
      match List.nth_error trace (Abstraction.snap_pc ks) with
      | Some instr =>
          WFDrivenPrecondition ks instr /\
          WFDrivenRun fuel' trace (Abstraction.kami_step ks instr)
      | None => True
      end
  end
     : nat ->
       list VMStep.VMStep.vm_instruction -> Abstraction.KamiSnapshot -> Prop
