counter_witness =
fun u v : nat =>
{|
  VMState.wc_same_00 := S u;
  VMState.wc_diff_00 := S v;
  VMState.wc_same_01 := 1;
  VMState.wc_diff_01 := 1;
  VMState.wc_same_10 := 1;
  VMState.wc_diff_10 := 0;
  VMState.wc_same_11 := 1;
  VMState.wc_diff_11 := 1
|}
     : nat -> nat -> VMState.WitnessCounts

witness_counter_equality_test
     : forall u v : nat,
       VMStep.VMStep.column_contractive_check_witness (counter_witness u v) =
       true <-> u = v
