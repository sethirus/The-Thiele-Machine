Inductive normalization_rule_kind : Set :=
    NormalizeStart : normalization_rule_kind
  | NormalizeScan : normalization_rule_kind
  | NormalizeEmit : normalization_rule_kind
  | NormalizeCommit : normalization_rule_kind.
normalization_kind_rule =
fun kind : normalization_rule_kind =>
match kind with
| NormalizeStart => NormalizationStart.normalization_start_rule
| NormalizeScan => NormalizationSteps.normalization_scan_rule
| NormalizeEmit => NormalizationSteps.normalization_emit_rule
| NormalizeCommit => NormalizationSteps.normalization_commit_rule
end
     : normalization_rule_kind ->
       Struct.Attribute (Syntax.Action Syntax.Void)

Arguments normalization_kind_rule kind
normalization_kind_phase =
fun kind : normalization_rule_kind =>
match kind with
| NormalizeStart => 5
| NormalizeScan => 8
| NormalizeEmit => 9
| NormalizeCommit => 11
end
     : normalization_rule_kind -> nat

Arguments normalization_kind_phase kind
normalization_action_enabled =
fun (old : Semantics.RegsT) (kind : normalization_rule_kind) =>
exists (u : Semantics.UpdatesT) (cs : Semantics.MethsT),
  Semantics.SemAction old
    (Struct.attrType (normalization_kind_rule kind) Syntax.type) u cs Word.WO
     : Semantics.RegsT -> normalization_rule_kind -> Prop

Arguments normalization_action_enabled old kind
normalization_register_bit_value =
fun r : {x : Syntax.FullKind & Syntax.fullType Syntax.type x} =>
let (x, w) := r in
match x as x0 return (Syntax.fullType Syntax.type x0 -> nat) with
| Syntax.SyntaxKind k =>
    match
      k as k0
      return (Syntax.fullType Syntax.type (Syntax.SyntaxKind k0) -> nat)
    with
    | Syntax.Bool =>
        fun _ : Syntax.fullType Syntax.type (Syntax.SyntaxKind Syntax.Bool)
        => 0
    | Syntax.Bit n =>
        fun
          w0 : Syntax.fullType Syntax.type (Syntax.SyntaxKind (Syntax.Bit n))
        => Word.wordToNat w0
    | Syntax.Vector k0 n =>
        fun
          _ : Syntax.fullType Syntax.type
                (Syntax.SyntaxKind (Syntax.Vector k0 n)) => 0
    | @Syntax.Struct n t =>
        fun
          _ : Syntax.fullType Syntax.type
                (Syntax.SyntaxKind (Syntax.Struct t)) => 0
    | Syntax.Array k0 n =>
        fun
          _ : Syntax.fullType Syntax.type
                (Syntax.SyntaxKind (Syntax.Array k0 n)) => 0
    end
| @Syntax.NativeKind t c =>
    fun _ : Syntax.fullType Syntax.type (Syntax.NativeKind c) => 0
end w
     : {x : Syntax.FullKind & Syntax.fullType Syntax.type x} -> nat

Arguments normalization_register_bit_value r
normalization_optional_bit_value =
fun r : option {x : Syntax.FullKind & Syntax.fullType Syntax.type x} =>
match r with
| Some v => normalization_register_bit_value v
| None => 0
end
     : option {x : Syntax.FullKind & Syntax.fullType Syntax.type x} -> nat

Arguments normalization_optional_bit_value r
