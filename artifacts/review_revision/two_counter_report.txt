two_counter_witness =
fun u v a b : nat =>
{|
  VMState.wc_same_00 := S u;
  VMState.wc_diff_00 := S v;
  VMState.wc_same_01 := S a;
  VMState.wc_diff_01 := S b;
  VMState.wc_same_10 := 1;
  VMState.wc_diff_10 := 0;
  VMState.wc_same_11 := 1;
  VMState.wc_diff_11 := 1
|}
     : nat -> nat -> nat -> nat -> VMState.WitnessCounts

Arguments two_counter_witness (u v a b)%nat_scope
two_counter_update_result =
fun (s : VMState.VMState) (wc : VMState.WitnessCounts) =>
{|
  VMState.vm_graph := VMState.vm_graph s;
  VMState.vm_csrs := VMState.vm_csrs s;
  VMState.vm_regs := VMState.vm_regs s;
  VMState.vm_mem := VMState.vm_mem s;
  VMState.vm_pc := S (VMState.vm_pc s);
  VMState.vm_mu := VMState.vm_mu s;
  VMState.vm_mu_tensor := VMState.vm_mu_tensor s;
  VMState.vm_err := VMState.vm_err s;
  VMState.vm_logic_acc := VMState.vm_logic_acc s;
  VMState.vm_mstatus := VMState.vm_mstatus s;
  VMState.vm_witness := wc;
  VMState.vm_certified := VMState.vm_certified s
|}
     : VMState.VMState -> VMState.WitnessCounts -> VMState.VMState

Arguments two_counter_update_result s wc
two_counter_next =
fun (u v a b : nat) (second different : bool) =>
if second
then
 if different
 then two_counter_witness u v a (S b)
 else two_counter_witness u v (S a) b
else
 if different
 then two_counter_witness u (S v) a b
 else two_counter_witness (S u) v a b
     : nat -> nat -> nat -> nat -> bool -> bool -> VMState.WitnessCounts
