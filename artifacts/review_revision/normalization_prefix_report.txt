selected_prefix =
fun (src dst : NormalizationSteps.PairTable) (b i e : nat) =>
List.map (NormalizationLoop.table_pair src dst)

       nat -> nat -> nat -> list NormalizationLoop.coupling_pair

normalization_prefix_invariant =
fun (rawsrc rawdst src dst : NormalizationSteps.PairTable) (b i out e : nat)
=>
(b <= out <= i /\ i <= e <= 16) /\
NormalizationLoop.table_slice src dst b (out - b) =
selected_prefix rawsrc rawdst b i e /\
(forall k : nat,
 i <= k < e ->
 NormalizationLoop.table_pair src dst k =
 NormalizationLoop.table_pair rawsrc rawdst k)
     : NormalizationSteps.PairTable ->
       NormalizationSteps.PairTable ->
       NormalizationSteps.PairTable ->
       NormalizationSteps.PairTable -> nat -> nat -> nat -> nat -> Prop
