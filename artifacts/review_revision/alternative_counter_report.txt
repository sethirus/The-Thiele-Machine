diagonal_counter_witness =
fun u v a b : nat =>
{|
  VMState.wc_same_00 := S u;
  VMState.wc_diff_00 := S v;
  VMState.wc_same_01 := 1;
  VMState.wc_diff_01 := 1;
  VMState.wc_same_10 := 1;
  VMState.wc_diff_10 := 1;
  VMState.wc_same_11 := S a;
  VMState.wc_diff_11 := S b
|}
     : nat -> nat -> nat -> nat -> VMState.WitnessCounts

Arguments diagonal_counter_witness (u v a b)%nat_scope
diagonal_sos_condition =
fun u v a b : nat =>
let x := VMStep.VMStep.chsh_d_z (S u) (S v) in
let nx := VMStep.VMStep.chsh_n_z (S u) (S v) in
let y := VMStep.VMStep.chsh_d_z (S a) (S b) in
let ny := VMStep.VMStep.chsh_n_z (S a) (S b) in
BinInt.Z.le
  (BinInt.Z.add (BinInt.Z.mul (BinInt.Z.mul (BinInt.Z.mul x x) ny) ny)
     (BinInt.Z.mul (BinInt.Z.mul (BinInt.Z.mul y y) nx) nx))
  (BinInt.Z.mul (BinInt.Z.mul (BinInt.Z.mul nx nx) ny) ny)
     : nat -> nat -> nat -> nat -> Prop
